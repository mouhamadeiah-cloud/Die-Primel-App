function loadFromFirebase() { // Stammdaten laden db.ref('data').once('value', (snapshot) => { const data = snapshot.val(); if (data && data.categories) { categories = data.categories || []; products = data.products || []; extras = data.extras || []; zutaten = data.zutaten || []; sales = data.sales || []; } ensureOrders(); renderCats(); if (categories.length > 0) renderProds(categories[0].id); }); // LISTENER für aktive Bestellungen db.ref('active_orders').on('value', (snapshot) => { const activeData = snapshot.val(); const newOrders = []; if (activeData) { Object.entries(activeData).forEach(([tableNum, batches]) => { // Prüfen ob wir diese Bestellung schon haben const existingOrder = orders.find(o => o.table == tableNum && o.status === 'open'); if (existingOrder) { // Existierende Bestellung aktualisieren existingOrder.batches = []; Object.entries(batches).forEach(([batchId, batch]) => { existingOrder.batches.push({ batchId: batch.batchId || batchId, time: batch.time, date: batch.date, items: batch.items || [], status: batch.status || 'new' }); }); newOrders.push(existingOrder); } else { // Neue Bestellung erstellen const order = { id: Date.now() + Math.random(), table: tableNum, status: 'open', batches: [] }; Object.entries(batches).forEach(([batchId, batch]) => { order.batches.push({ batchId: batch.batchId || batchId, time: batch.time, date: batch.date, items: batch.items || [], status: batch.status || 'new' }); }); newOrders.push(order); } }); } // orders Array aktualisieren orders = newOrders; localStorage.setItem('orders', JSON.stringify(orders)); // UI aktualisieren wenn Tische-Modal offen if (document.getElementById('modal-tables').style.display === 'block') { openTables(); } }); }
