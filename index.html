// ========== FUNKTION ZUM AKTUALISIEREN DER FREIEN TISCHE ========== function updateFreeTablesDisplay() { // Lade alle aktiven Bestellungen von Firebase db.ref('active_orders').once('value', (snapshot) => { const occupiedTables = []; // Sammle alle besetzten Tische snapshot.forEach((tableSnapshot) => { const tableNum = tableSnapshot.key; if (tableNum !== 'takeaway') { occupiedTables.push(tableNum); } }); // Aktualisiere die Tisch-Buttons im Grid const tableButtons = document.querySelectorAll('#table-grid .table-btn'); tableButtons.forEach(btn => { const tableNum = btn.textContent; if (occupiedTables.includes(tableNum)) { btn.style.display = 'none'; // Besetzte Tische verstecken } else { btn.style.display = 'inline-block'; // Freie Tische zeigen } }); // Prüfe ob es freie Tische gibt const freeTablesCount = Array.from(tableButtons).filter(btn => btn.style.display !== 'none').length; if (freeTablesCount === 0) { alert('Keine freien Tische verfügbar! Nur Takeaway möglich.'); } }); } // ========== TABLE SELECTION (مع التأثير البصري) ========== function selectTable(num) { // إزالة الكلاس selected من جميع الأزرار document.querySelectorAll('#table-grid .table-btn').forEach(btn => { btn.classList.remove('selected'); }); // إضافة الكلاس selected للزر الحالي event.target.classList.add('selected'); // حفظ الرقم المختار selectedTable = num; // عرض الرقم في حقل الإدخال document.getElementById('selected-table-display').innerText = `Tisch ${num}`; console.log("Tisch ausgewählt:", num); // للتأكد من العمل } function selectTakeaway() { // إزالة الكلاس selected من جميع الأزرار document.querySelectorAll('#table-grid .table-btn').forEach(btn => { btn.classList.remove('selected'); }); // حفظ Takeaway selectedTable = 'takeaway'; // عرض Takeaway في حقل الإدخال document.getElementById('selected-table-display').innerText = '🥡 Takeaway'; console.log("Takeaway ausgewählt"); // للتأكد من العمل } function confirmTableSelection() { if (!selectedTable) { alert("Bitte wählen Sie einen Tisch oder Takeaway!"); return; } activeTable = selectedTable; closeModal('modal-table-grid'); confirmOrder(); } // ========== initiateSend (بدون تغيير) ========== function initiateSend() { if(cartNew.length === 0) { alert("Warenkorb leer!"); return; } if(activeTable) { confirmOrder(); } else { updateFreeTablesDisplay(); // إعادة تعيين الاختيار السابق selectedTable = null; document.querySelectorAll('#table-grid .table-btn').forEach(btn => { btn.classList.remove('selected'); }); document.getElementById('selected-table-display').innerText = 'Kein Tisch gewählt'; openModal('modal-table-grid'); } }
